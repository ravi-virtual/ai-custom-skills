# DATABASE.md — Database Engineering Reference

## Database: MongoDB 7.x via Mongoose 8.x

---

## Connection Setup
```javascript
// src/config/database.js
'use strict';

const mongoose = require('mongoose');
const { config } = require('./constants');

const connectDB = async () => {
  try {
    const conn = await mongoose.connect(config.MONGO_URI, {
      // Mongoose 8.x defaults: no options needed for basic connection
    });
    console.log(`MongoDB Connected: ${conn.connection.host}`);
  } catch (err) {
    console.error(`MongoDB connection failed: ${err.message}`);
    process.exit(1); // Fail fast on DB connection failure
  }
};

const disconnectDB = async () => {
  await mongoose.disconnect();
};

module.exports = { connectDB, disconnectDB };
```

## Schema Design Principles

### Non-Negotiable Fields (Every Collection)
```javascript
{
  _id: { type: String, default: uuidv4 },   // UUID, not ObjectId
  isDeleted: { type: Boolean, default: false }, // Soft delete
}
// Plus:
{ timestamps: true, _id: false }            // Auto createdAt + updatedAt
```

### Schema Options Template
```javascript
const schema = new mongoose.Schema(
  {
    _id: { type: String, default: uuidv4 },
    // ... fields
    isDeleted: { type: Boolean, default: false },
  },
  {
    timestamps: true,     // Adds createdAt + updatedAt
    _id: false,           // Prevents Mongoose from adding its own ObjectId
    toJSON: {
      transform(doc, ret) {
        delete ret.__v;
        delete ret.password; // If applicable
        return ret;
      }
    }
  }
);
```

## CRUD Operation Patterns

### Create
```javascript
// Single document
const user = await User.create({
  name: 'John',
  email: 'john@example.com',
  // _id auto-generated by uuidv4 default
});

// With explicit ID (for idempotent creation)
const user = await User.create({
  _id: providedUUID,
  name: 'John',
  email: 'john@example.com',
});
```

### Read
```javascript
// Single document by ID
const user = await User.findOne({ _id: id, isDeleted: false }).lean();

// List with pagination
const page = parseInt(req.query.page) || 1;
const limit = parseInt(req.query.limit) || 20;
const skip = (page - 1) * limit;

const [users, totalCount] = await Promise.all([
  User.find({ isDeleted: false })
    .select('name email role createdAt')
    .sort({ createdAt: -1 })
    .skip(skip)
    .limit(limit)
    .lean(),
  User.countDocuments({ isDeleted: false })
]);

// Search by field
const user = await User.findOne({ email: email.toLowerCase(), isDeleted: false }).lean();

// Admin: find all including soft-deleted
const allUsers = await User.find({}).lean(); // Only for admin data management
```

### Update
```javascript
// Find and update (returns updated document)
const updatedUser = await User.findOneAndUpdate(
  { _id: id, isDeleted: false },
  { $set: { name: newName } },
  { new: true }  // Return the updated document, not the original
).lean();

// Direct update (does not return document, more efficient)
await User.updateOne(
  { _id: id, isDeleted: false },
  { $set: { name: newName } }
);
```

### Delete (Soft Delete ONLY)
```javascript
// Soft delete — NEVER use deleteOne/deleteMany on user data
const deletedUser = await User.findOneAndUpdate(
  { _id: id, isDeleted: false },
  { $set: { isDeleted: true } },
  { new: true }
).lean();

if (!deletedUser) {
  // Either doesn't exist or already deleted
  return sendError(res, { statusCode: ResponseCodes.NOT_FOUND, message: 'Not found' });
}
```

## Indexing Reference

### Define Indexes on Schema
```javascript
// Single field index
userSchema.index({ email: 1 }, { unique: true });

// Compound index (for queries using both fields)
userSchema.index({ isDeleted: 1, role: 1 });

// Time-sorted queries
userSchema.index({ isDeleted: 1, createdAt: -1 });

// Text search index
userSchema.index({ name: 'text', email: 'text' });
```

### Index Guidelines
- Index fields used in `.find()`, `.findOne()`, or `.countDocuments()` conditions
- Index fields used in `.sort()`
- Compound indexes: put equality conditions first, then range, then sort
- Every unique field should have a unique index
- Don't over-index — each index slows writes

## Transaction Support
```javascript
// For operations that must be atomic across multiple collections
const session = await mongoose.startSession();
try {
  session.startTransaction();
  
  await User.create([userData], { session });
  await Profile.create([profileData], { session });
  
  await session.commitTransaction();
} catch (err) {
  await session.abortTransaction();
  throw err;
} finally {
  await session.endSession();
}
```

## Data Model Reference

### User Model
```javascript
{
  _id: String (UUID),
  name: String,
  email: String (unique, lowercase),
  password: String (bcrypt hashed, select: false),
  role: 'admin' | 'user',
  isDeleted: Boolean,
  createdAt: Date,
  updatedAt: Date
}
```

### Profile Model
```javascript
{
  _id: String (UUID),
  userId: String (ref to User._id),
  name: String,
  // ... other profile fields,
  isDeleted: Boolean,
  createdAt: Date,
  updatedAt: Date
}
```

## Query Optimization Checklist
Before writing any DB query:
- [ ] Included `{ isDeleted: false }` in conditions
- [ ] Used `.lean()` for read queries
- [ ] Used `.select()` to limit returned fields
- [ ] Used `.limit()` for list queries
- [ ] Queries use indexed fields in conditions
- [ ] Avoided `$where` (JavaScript execution, slow and injection risk)
- [ ] Used `Promise.all()` for parallel independent queries

## MongoDB Error Handling
```javascript
// Common MongoDB errors to handle

try {
  await User.create(userData);
} catch (err) {
  if (err.code === 11000) {
    // Duplicate key error (unique index violation)
    const field = Object.keys(err.keyPattern)[0];
    return sendError(res, {
      statusCode: ResponseCodes.CONFLICT,
      message: `${field} already exists`,
      code: ErrorCodes.DUPLICATE_ERROR,
    });
  }
  // Other errors fall through to global handler
  throw err;
}
```
